How to add SoundWave Auth to ANY website

Copy and paste this script into the <head> or body of any HTML page:

<!-- 1. Load SoundWave SDK -->
<script src="c:/SoundWave.Login.io/soundwave-auth-sdk.js"></script>

<!-- 2. Initialize and trigger authentication modal -->
<script>
  SoundWaveAuth.init({
    appId: 'my-website-1',
    appName: 'My Awesome Website',
    onSuccess: (user) => {
      console.log('User logged in:', user.username, user.role);
      alert('Welcome ' + user.username + '!');
    },
    onLogout: () => {
      console.log('User logged out');
    }
  });

  // Attach login button click event
  document.getElementById('my-login-btn').addEventListener('click', () => {
    SoundWaveAuth.openModal();
  });
</script>
